import os
import sys

try:
    from .console import Colors, ConsoleUI, _BANNER_ART, _HAS_RICH
    if _HAS_RICH:
        from rich.console import Console
        from rich.text import Text
        _HAS_RICH_CONSOLE = True
    else:
        _HAS_RICH_CONSOLE = False
except Exception:
    _HAS_RICH = False
    _HAS_RICH_CONSOLE = False

    class Colors:
        RED = '\033[91m'
        GREEN = '\033[92m'
        YELLOW = '\033[93m'
        CYAN = '\033[96m'
        RESET = '\033[0m'
        BOLD = '\033[1m'
        UNDERLINE = '\033[4m'
        MAGENTA = '\033[95m'
        WHITE = '\033[97m'
        BLUE = '\033[94m'


_PUTTY_BANNER = f"""{Colors.CYAN}
  _____           _             _   _               _       
 |  __ \\         | |           | | (_)             | |      
 | |__) |   _ _ __| |_ ___ _ __ | |_| |__   ___ _ __| |_ ___ 
 |  ___/ | | | '__| __/ _ \\ '__ \\| __| '_ \\ / _ \\ '__| __/ __|
 | |   | |_| | |  | ||  __/ | | | |_| | | |  __/ |  | |_\\__ \\
 |_|    \\__,_|_|   \\__\\___|_| |_|_\\__|_| |_|\\___|_|   \\__|___/{Colors.RESET}
{Colors.YELLOW}  PuTTY Terminal Engine v0.78-SSH{Colors.RESET}
{Colors.CYAN}  Secure Shell Protocol Client{Colors.RESET}
"""


class PuttyUI:
    """PuTTY-style terminal UI that renders simple text-based menus."""

    PROMPT_USER = "user"

    def __init__(self, console=None):
        if _HAS_RICH_CONSOLE:
            self._console = console or Console(force_terminal=True)
        else:
            self._console = console
        self._username = self.PROMPT_USER
        self._host = "localhost"

    def clear(self):
        if self._console:
            self._console.clear()
        else:
            os.system('cls' if os.name == 'nt' else 'clear')

    def _print(self, text="", style=""):
        if self._console:
            from rich.text import Text as _Text
            self._console.print(_Text(text, style=style) if not style else text)
        else:
            print(text)

    def _input(self, prompt=""):
        if self._console:
            return self._console.input(prompt)
        else:
            return input(prompt)

    def rule_header(self, title, color="cyan"):
        line = "=" * 60
        print(f"\n{Colors.CYAN}{line}{Colors.RESET}")
        print(f"{Colors.BOLD}{title}{Colors.RESET}")
        print(f"{Colors.CYAN}{line}{Colors.RESET}\n")

    def info(self, msg):
        print(f"{Colors.CYAN}[*]{Colors.RESET} {msg}")

    def success(self, msg):
        print(f"{Colors.GREEN}[+]{Colors.RESET} {msg}")

    def warning(self, msg):
        print(f"{Colors.YELLOW}[!]{Colors.RESET} {msg}")

    def error(self, msg):
        print(f"{Colors.RED}[X]{Colors.RESET} {msg}")

    def pause(self, msg="Press Enter to continue..."):
        print(f"{Colors.YELLOW}{msg}{Colors.RESET}")
        try:
            self._input()
        except (KeyboardInterrupt, EOFError):
            pass

    def panel(self, content, title="", border_style="cyan",
              box_style=None, padding=None, subtitle=None):
        color = getattr(Colors, border_style.upper(), Colors.CYAN) if isinstance(border_style, str) else Colors.CYAN
        print(f"\n{color}+{'-' * 58}+{Colors.RESET}")
        if title:
            print(f"{color}| {Colors.BOLD}{title}{Colors.RESET}{' ' * (56 - len(title))}{color}|{Colors.RESET}")
        print(f"{color}|{' ' * 58}|{Colors.RESET}")
        for line in str(content).split('\n'):
            print(f"{color}|{Colors.RESET} {line}{' ' * (57 - len(line))}{color}|{Colors.RESET}")
        if subtitle:
            print(f"{color}|{Colors.RESET} {subtitle}{' ' * (58 - len(subtitle))}{color}|{Colors.RESET}")
        print(f"{color}+{'-' * 58}+{Colors.RESET}")

    def section(self, title, color="cyan", subtitle=None):
        color_code = getattr(Colors, color.upper(), Colors.CYAN) if isinstance(color, str) else Colors.CYAN
        print(f"\n{color_code}{'-' * 60}{Colors.RESET}")
        print(f"{color_code}{title}{Colors.RESET}")
        if subtitle:
            print(f"{Colors.YELLOW}{subtitle}{Colors.RESET}")
        print(f"{color_code}{'-' * 60}{Colors.RESET}")

    def keyval_table(self, title, rows, border_style="cyan"):
        color = getattr(Colors, border_style.upper(), Colors.CYAN) if isinstance(border_style, str) else Colors.CYAN
        if title:
            print(f"\n{color}{title}{Colors.RESET}")
        max_label = max(len(label) for label, _ in rows) if rows else 20
        for label, value in rows:
            print(f"  {Colors.BOLD}{label.ljust(max_label)}{Colors.RESET} : {value}")

    def result_table(self, title, headers, rows, border_style="cyan"):
        color = getattr(Colors, border_style.upper(), Colors.CYAN) if isinstance(border_style, str) else Colors.CYAN
        if title:
            print(f"\n{color}{title}{Colors.RESET}")
        print(f"{color}{' | '.join(str(h) for h in headers)}{Colors.RESET}")
        print(f"{color}{'-' * (len(' | '.join(str(h) for h in headers)))}{Colors.RESET}")
        for row in rows:
            cells = list(row) if isinstance(row, (list, tuple)) else [row]
            print(f"{color}{' | '.join(str(c) for c in cells)}{Colors.RESET}")

    def prompt_input(self, prompt, default=None):
        if default is not None:
            line = f"{Colors.YELLOW}{prompt} (default: {default}) -> {Colors.RESET}"
        else:
            line = f"{Colors.YELLOW}{prompt} -> {Colors.RESET}"
        value = self._input(line).strip()
        if value == "" and default is not None:
            return default
        return value

    def banner(self):
        print(_PUTTY_BANNER)
        print(f"{Colors.GREEN}  MAINFRAME // SEC-OPS TERMINAL MULTITOOL{Colors.RESET}")
        print(f"{Colors.CYAN}  DEPLOYMENT SPECIFICATION RELEASE v5.90 // 40-IN-1 TOOL PLATFORM{Colors.RESET}")
        print()

    def main_menu(self):
        print(f"\n{Colors.CYAN}============================================================{Colors.RESET}")
        print(f"{Colors.BOLD}  MAIN SYSTEM DIRECTORY CORE{Colors.RESET}")
        print(f"{Colors.CYAN}============================================================{Colors.RESET}")
        print(f"  {Colors.CYAN}[01]{Colors.RESET} Sub-Directory 01 // Network Infrastructure & Endpoint Recon Cores")
        print(f"  {Colors.CYAN}[02]{Colors.RESET} Sub-Directory 02 // External OSINT & Target Record Profilers")
        print(f"  {Colors.GREEN}[03]{Colors.RESET} Sub-Directory 03 // Local Data Traffic Monitors, Audits & Utilities")
        print(f"  {Colors.CYAN}[04]{Colors.RESET} Sub-Directory 04 // Advanced Infrastructure Audits & Integrity Cores")
        print(f"  {Colors.RED}[05]{Colors.RESET} Sub-Directory 05 // Attack Vectors & Framework Interfaces [SHELL BASELINE]")
        print(f"  {Colors.BLUE}[06]{Colors.RESET} Sub-Directory 06 // Discord Token, Webhook & Server Tools")
        print(f"  {Colors.WHITE}[07]{Colors.RESET} Sub-Directory 07 // Dox Tools & OSINT Suites")
        print(f"  {Colors.RED}[08]{Colors.RESET} Sub-Directory 08 // Payload Builder & Generation")
        print(f"  {Colors.GREEN}[09]{Colors.RESET} Sub-Directory 09 // Roblox Intelligence & Cookie Tools")
        print(f"  {Colors.RED}[10]{Colors.RESET} Terminate Active Mainframe Operator Control Session")
        print(f"{Colors.CYAN}============================================================{Colors.RESET}")

    def submenu(self, title, items, back_text=None):
        print(f"\n{Colors.CYAN}{title}{Colors.RESET}")
        print(f"{Colors.CYAN}{'-' * 60}{Colors.RESET}")
        for item in items:
            if isinstance(item, (list, tuple)) and len(item) >= 3:
                key, label, color = item[0], item[1], item[2]
                color_code = getattr(Colors, color.upper(), Colors.WHITE) if isinstance(color, str) else Colors.WHITE
                print(f"  {Colors.BOLD}[{key}]{Colors.RESET} {color_code}{label}{Colors.RESET}")
            elif isinstance(item, (list, tuple)) and len(item) == 2:
                key, label = item[0], item[1]
                print(f"  [{key}] {label}")
            else:
                print(f"  {item}")
        if back_text:
            print(f"\n{Colors.YELLOW}{back_text}{Colors.RESET}")

    def directory_menu(self, title, items, header_color="cyan"):
        header_color_code = getattr(Colors, header_color.upper(), Colors.CYAN) if isinstance(header_color, str) else Colors.CYAN
        print(f"\n{Colors.BOLD}{header_color_code}[{title}]{Colors.RESET}")
        print(f"{Colors.CYAN}{'-' * 60}{Colors.RESET}")
        if not items:
            return
        max_name = max(len(item[1]) for item in items)
        for item in items:
            raw_key = item[0]
            key = f"{int(raw_key):02d}" if str(raw_key).isdigit() else str(raw_key)
            name = item[1]
            desc = item[2] if len(item) > 2 else ""
            color = item[3] if len(item) > 3 else header_color
            color_code = getattr(Colors, color.upper(), header_color_code) if isinstance(color, str) else header_color_code
            print(f"  {Colors.BOLD}{color_code}[{key}]{Colors.RESET} {color_code}{name.ljust(max_name)}{Colors.RESET} - {desc}")

    def prompt_input(self, prompt, default=None):
        if default is not None:
            line = f"{Colors.YELLOW}{prompt} (default: {default}) -> {Colors.RESET}"
        else:
            line = f"{Colors.YELLOW}{prompt} -> {Colors.RESET}"
        value = self._input(line).strip()
        if value == "" and default is not None:
            return default
        return value
